################################################################################
# SPARTA Couette Flow - Bird Chapter 12.2 Argon Viscosity Measurement
# FULL BIRD SIMULATION: Matches DSMC1.FOR exactly
# 
# Expected: μ = 2.117×10⁻⁵ Pa·s at 273K (Bird Section 12.2)
# 
# FULL BIRD SIMULATION PARAMETERS:
#   Phase 1: 50,000 steps transient (discarded) = 0.125 seconds
#   Phase 2: 100,000 steps steady state (saved) = 0.25 seconds
#   Total: 150,000 steps = 0.375 seconds
################################################################################

################################################################################
# INITIALIZATION
################################################################################

seed                12345 
dimension           2     
global              gridcut 0.0 comm/sort yes

################################################################################
# BOUNDARY CONDITIONS
################################################################################

boundary            p s p
surf_collide        1 diffuse 273.0 1.0
surf_collide        2 diffuse 273.0 1.0 translate 300.0 0 0
bound_modify        ylo collide 1
bound_modify        yhi collide 2

################################################################################
# DOMAIN GEOMETRY
################################################################################

create_box          0 0.01 0 1.0 -0.0005 0.0005

################################################################################
# GRID SETUP
################################################################################

create_grid         1 200 1 
balance_grid        rcb cell

################################################################################
# GAS SPECIES AND MIXTURE
################################################################################

species             ar.species Ar
mixture             gas Ar temp 273.0 vstream 0.0 0.0 0.0

################################################################################
# PARTICLE WEIGHTING AND INITIALIZATION
################################################################################

global              fnum 1.4e14 nrho 1.4e20 
create_particles    gas n 10000

################################################################################
# COLLISION MODEL
################################################################################

collide             vss gas ar.vss 

################################################################################
# COMPUTE: GRID PROPERTIES
################################################################################

compute             gcomp grid all all u v w massrho nrho temp
compute             gtcomp thermal/grid all all temp 
compute             gpcomp thermal/grid all all press
compute             hfcomp eflux/grid all species heatx heaty heatz
compute             mfcomp pflux/grid all species momxy

################################################################################
# TIME-AVERAGING FIXES (will be reset after transient)
################################################################################

fix                 gfix ave/grid all 1 500 500 c_gcomp[*]
fix                 gtfix ave/grid all 1 500 500 c_gtcomp[*] 
fix                 gpfix ave/grid all 1 500 500 c_gpcomp[*]
fix                 hfix ave/grid all 1 500 500 c_hfcomp[*]
fix                 mfix ave/grid all 1 500 500 c_mfcomp[*]

################################################################################
# OUTPUT FILES (will be enabled after transient)
################################################################################

dump                gdump grid all 500 grid.*.dat id yc f_gfix[*] f_gtfix[*] f_gpfix[*] f_hfix[*] f_mfix[*]

################################################################################
# STATISTICS OUTPUT
################################################################################

stats               500 
stats_style         step cpu np nattempt ncoll nscoll nscheck

################################################################################
# TIME INTEGRATION
################################################################################

timestep            2.5e-6

################################################################################
# PHASE 1: TRANSIENT PERIOD (DISCARD DATA)
################################################################################

undump              gdump

print               "======================================================================"
print               "FULL BIRD CHAPTER 12.2 SIMULATION"
print               "======================================================================"
print               "PHASE 1: Running transient period (50000 steps)"
print               "         No output files will be generated"
print               "         Simulation time: 0.125 seconds"
print               "======================================================================"

run                 50000

################################################################################
# PHASE 2: STEADY STATE PERIOD (SAVE DATA)
################################################################################

print               "======================================================================"
print               "PHASE 2: Resetting statistics for steady state"
print               "======================================================================"

unfix               gfix
unfix               gtfix  
unfix               gpfix
unfix               hfix
unfix               mfix

fix                 gfix ave/grid all 1 500 500 c_gcomp[*] ave running
fix                 gtfix ave/grid all 1 500 500 c_gtcomp[*] ave running
fix                 gpfix ave/grid all 1 500 500 c_gpcomp[*] ave running
fix                 hfix ave/grid all 1 500 500 c_hfcomp[*] ave running
fix                 mfix ave/grid all 1 500 500 c_mfcomp[*] ave running

dump                gdump grid all 50000 grid.*.dat id yc f_gfix[*] f_gtfix[*] f_gpfix[*] f_hfix[*] f_mfix[*]

print               "         Running steady state (100000 steps)"
print               "         Simulation time: 0.25 seconds"
print               "======================================================================"

run                 100000

################################################################################
# SIMULATION COMPLETE
################################################################################

print               "======================================================================"
print               "SIMULATION COMPLETE - FULL BIRD CHAPTER 12.2"
print               "======================================================================"
print               "Total timesteps: 150000 (matching Bird DSMC1.FOR exactly)"
print               "Total simulation time: 0.375 seconds"
print               "Transient discarded: 50000 steps (0.125 s)"
